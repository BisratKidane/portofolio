```
function MemberRows({ members, self, isAdmin, actingUserId, isDerived, onEdit, onDelete }) {
  return (
    <Stack spacing={2} sx={{ mt: 2 }}>
      {members.map((member) => (
        <MemberCard key={member.id} member={member} isAdmin={isAdmin} actingUserId={actingUserId}
          isSelf={member.id === self.id} isDerived={isDerived} onEdit={onEdit} onDelete={onDelete} />
      ))}
    </Stack>
  );
}

export default function RelationshipGroupedPanel({ scope, isAdmin, actingUserId, onAddRelative, onEdit, onDelete }) {
  const rowProps = { self, isAdmin, actingUserId, onEdit, onDelete };
  return (
    <Paper ...>
      <MemberCard member={self} isAdmin={isAdmin} actingUserId={actingUserId} isSelf isDerived={false} onEdit={onEdit} onDelete={onDelete} />
      ...
      <MemberRows members={parents} isDerived={false} {...rowProps} />
      <MemberRows members={spouses} isDerived={false} {...rowProps} />
      <MemberRows members={children} isDerived={false} {...rowProps} />
      <MemberRows members={siblings} isDerived {...rowProps} />
    </Paper>
  );
}
```
